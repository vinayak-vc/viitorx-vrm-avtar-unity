BLOCK 1 — §14 performance + allocation, old vs new
--------------------------------------------------------------------------------------------------
var sw = new System.Diagnostics.Stopwatch();
int iterations = 200000;
var lm = new Vector3[33];
lm[11] = new Vector3( 0.18f, -0.50f, 0f);     lm[12] = new Vector3(-0.18f, -0.50f, 0f);
lm[13] = new Vector3( 0.42f, -0.36f, -0.10f); lm[14] = new Vector3(-0.42f, -0.36f, -0.10f);
lm[15] = new Vector3( 0.55f, -0.18f, -0.22f); lm[16] = new Vector3(-0.55f, -0.18f, -0.22f);
long allocBefore = System.GC.GetTotalMemory(true);
sw.Restart();
float sink = 0f;
for (int i = 0; i < iterations; i++) {
    var arms = VirtualMirror.Retargeting.Kalidokit.KalidokitArmSolver.Solve(lm);
    var qa = VirtualMirror.Retargeting.Kalidokit.KMath.ThreeEulerToUnity(arms.UpperArmLeft,  Vector3.one, 2);
    var qb = VirtualMirror.Retargeting.Kalidokit.KMath.ThreeEulerToUnity(arms.LowerArmLeft,  Vector3.one, 2);
    var qc = VirtualMirror.Retargeting.Kalidokit.KMath.ThreeEulerToUnity(arms.UpperArmRight, Vector3.one, 2);
    var qd = VirtualMirror.Retargeting.Kalidokit.KMath.ThreeEulerToUnity(arms.LowerArmRight, Vector3.one, 2);
    sink += qa.w + qb.w + qc.w + qd.w;
}
sw.Stop();
double oldMs = sw.Elapsed.TotalMilliseconds;
long oldAlloc = System.GC.GetTotalMemory(false) - allocBefore;

var restL = VirtualMirror.Retargeting.ArmAimSolver.BuildRestBasis(Vector3.right, Vector3.right, Vector3.right, Vector3.up);
var restR = VirtualMirror.Retargeting.ArmAimSolver.BuildRestBasis(Vector3.left,  Vector3.left,  Vector3.right, Vector3.up);
var stL = new VirtualMirror.Retargeting.ArmRollState();
var stR = new VirtualMirror.Retargeting.ArmRollState();
allocBefore = System.GC.GetTotalMemory(true);
sw.Restart();
for (int i = 0; i < iterations; i++) {
    var a = VirtualMirror.Retargeting.ArmAimSolver.Solve(
        new Vector3(lm[12].x, -lm[12].y, lm[12].z), new Vector3(lm[14].x, -lm[14].y, lm[14].z),
        new Vector3(lm[16].x, -lm[16].y, lm[16].z), restL, ref stL, true);
    var b = VirtualMirror.Retargeting.ArmAimSolver.Solve(
        new Vector3(lm[11].x, -lm[11].y, lm[11].z), new Vector3(lm[13].x, -lm[13].y, lm[13].z),
        new Vector3(lm[15].x, -lm[15].y, lm[15].z), restR, ref stR, true);
    sink += a.UpperRig.w + b.UpperRig.w;
}
sw.Stop();
double newMs = sw.Elapsed.TotalMilliseconds;
long newAlloc = System.GC.GetTotalMemory(false) - allocBefore;
return new {
    iterations,
    old_us_per_frame = System.Math.Round(oldMs * 1000.0 / iterations, 3),
    new_us_per_frame = System.Math.Round(newMs * 1000.0 / iterations, 3),
    delta_us_per_frame = System.Math.Round((newMs - oldMs) * 1000.0 / iterations, 3),
    old_alloc_bytes = oldAlloc, new_alloc_bytes = newAlloc, sink
};

BLOCK 3 — §11 roll: 360 deg elbow-plane sweep + straighten/re-bend hysteresis
--------------------------------------------------------------------------------------------------
var st = new VirtualMirror.Retargeting.ArmRollState();
Vector3 axis = Vector3.right, pA = Vector3.up, pB = Vector3.Cross(Vector3.right, Vector3.up);
float prev = 0f, worstJump = 0f, worstDir = 0f; int flips = 0;
for (int d = 0; d < 360; d += 10) {
    float a = d*Mathf.Deg2Rad;
    Vector3 radial = pA*Mathf.Cos(a) + pB*Mathf.Sin(a);
    Vector3 fore = (axis*Mathf.Cos(60f*Mathf.Deg2Rad) + radial*Mathf.Sin(60f*Mathf.Deg2Rad)).normalized;
    Vector3[] aL = arm(shL, axis, fore);
    var r = VirtualMirror.Retargeting.ArmAimSolver.Solve(aL[0], aL[1], aL[2], restR, ref st, true);
    worstDir = Mathf.Max(worstDir, Mathf.Max(Vector3.Angle(mir(axis), r.UpperRig*restR.UpperDir),
                                             Vector3.Angle(mir(fore), r.LowerRig*restR.LowerDir)));
    if (d > 0) { float j = Mathf.Abs(Mathf.DeltaAngle(prev, r.RollDeg));
                 worstJump = Mathf.Max(worstJump, j); if (j > 90f) flips++; }
    prev = r.RollDeg;
}
// hysteresis: sweep the bend down through BendSinMin and back up
float[] bends = new float[]{ 60f, 30f, 10f, 3f, 0f, 3f, 10f, 30f, 60f };   // watch r.NormalSource

    hysteresis: ElbowPlane -> Held (at ~10 deg bend) -> ElbowPlane, roll continuous, 0.000 deg error
